# ============ NIEUWS OPHALEN ============
def haal_nieuws_op(aantal=5):
    try:
        rss_url = "https://feeds.nos.nl/nosnieuwsalgemeen"
        response = requests.get(rss_url, timeout=10)
        root = ET.fromstring(response.content)

        nieuws_items = []
        for item in root.findall('.//item')[:aantal]:
            title = item.find('title').text
            description = item.find('description').text
            link = item.find('link').text

            if description:
                text = html.unescape(description)
                text = re.sub(r'</(?:p|div|h[1-6]|br|li)>', '\n', text, flags=re.IGNORECASE)
                text = re.sub(r'<[^>]+>', '', text)
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                text = '\n\n'.join(lines)
            else:
                text = ""

            nieuws_items.append({
                'titel': title,
                'beschrijving': text,
                'link': link
            })

        return nieuws_items
    except Exception as e:
        st.error(f"Fout bij ophalen nieuws: {e}")
        return []

# ============ AI GEDACHTEN GENEREREN ============
def genereer_gedachte_ollama(nieuws_tekst, filosoof_key):
    try:
        filosoof_data = FILOSOFEN[filosoof_key]
        prompt = filosoof_data["prompt"]

        response = requests.post(
            "http://localhost:11434/api/generate",
            json={
                "model": "llama3.2",
                "prompt": f"{prompt}\n\nHet nieuws is: {nieuws_tekst}\n\nJouw analyse:",
                "stream": False,
                "options": {"temperature": 0.9}
            },
            timeout=180
        )

        resultaat = response.json()["response"]
        return resultaat.strip()
    except Exception as e:
        return f"❌ Fout met Ollama: {e}\n\n*Tip: Draait Ollama op de achtergrond?*"

def genereer_gedachte_openai(nieuws_tekst, filosoof_key):
    try:
        from openai import OpenAI
        client = OpenAI(api_key=OPENAI_API_KEY)
        filosoof_data = FILOSOFEN[filosoof_key]
        prompt = filosoof_data["prompt"]

        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": "Je bent een expert in de geschiedenis van de filosofie."},
                {"role": "user", "content": f"{prompt}\n\nHet nieuws is: {nieuws_tekst}\n\nJouw analyse:"}
            ],
            temperature=0.9,
            max_tokens=300
        )
        return response.choices[0].message.content.strip()
    except Exception as e:
        return f"❌ Fout met OpenAI: {e}"

def genereer_gedachte(nieuws_tekst, filosoof_key):
    if AI_PROVIDER == "openai":
        return genereer_gedachte_openai(nieuws_tekst, filosoof_key)
    else:
        return genereer_gedachte_ollama(nieuws_tekst, filosoof_key)
